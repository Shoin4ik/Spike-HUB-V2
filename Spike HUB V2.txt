local Cfg = {
    On = false, Radius = 250, Format = "Pairs", Bring = false, GatherRadius = 300,
    Farm = false, Quest = false, Raid = false, RaidType = "Flame", FarmRadius = 1500, Height = 10, Speed = 150,

local center = nil

local function active() return Cfg.Farm or Cfg.Quest or Cfg.Raid end
local function getRoot()
    local char = lp.Character
    return char and char:FindFirstChild("HumanoidRootPart")
end
local function notify(t, c)
    if Window then pcall(function() Window:Notify({ Title = t, Content = c }) end) end
end

local function mobMatch(mob)
    if Cfg.Quest and not Cfg.Farm and not Cfg.Raid and questMob then return mob.Name == questMob end
    return true
end

local function targets()
    local list = {}
    local root = getRoot()
    local folder = workspace:FindFirstChild("Enemies")
    if not (root and folder) then return list end
    for _, mob in ipairs(folder:GetChildren()) do
        local hum = mob:FindFirstChildOfClass("Humanoid")
        local head = mob:FindFirstChild("Head") or mob:FindFirstChild("HumanoidRootPart")
        if hum and hum.Health > 0 and head and (head.Position - root.Position).Magnitude <= Cfg.Radius then
            table.insert(list, { mob = mob, head = head })
            if #list >= 30 then break end
        end
    end
    return list
end

local function fireAura()
    local list = targets()
    if #list == 0 then return end
    local n = math.max(1, Cfg.Batch)
    for i = 1, #list, n do
        local chunk = {}
        for j = i, math.min(i + n - 1, #list) do
            if Cfg.Format == "Pairs" then
                chunk[#chunk + 1] = { list[j].mob, list[j].head }
            else
                chunk[#chunk + 1] = list[j].mob
            end
        end
        Atk:FireServer(0)
        Hit:FireServer(list[i].head, chunk, nil, token)
    end
end

-- AoE Kill Aura (also active while any auto farm is on)
task.spawn(function()
    while true do
        task.wait(0.15)
        if Cfg.On or active() then
            pcall(fireAura)
        end
    end
end)

-- Bring Mobs (also active while farming)
task.spawn(function()
    local lastSim = 0
    while true do
        task.wait(0.2)
        if Cfg.Bring or active() then
            pcall(function()
                local root = getRoot()
                local folder = workspace:FindFirstChild("Enemies")
                if not (root and folder) then return end
                if os.clock() - lastSim > 2 then
                    lastSim = os.clock()
                    if sethiddenproperty then pcall(sethiddenproperty, lp, "SimulationRadius", 1e9) end
                end
                local farming = active()
                local pos
                if farming and center then
                    pos = center
                else
                    pos = (root.CFrame * CFrame.new(0, 0, -8)).Position
                end
                local radius = Cfg.GatherRadius
                if farming then
                    local flat = Vector3.new(root.Position.X - pos.X, 0, root.Position.Z - pos.Z)
                    if flat.Magnitude > 40 then return end
                end
                local point = CFrame.new(pos)
                local n = 0
                for _, mob in ipairs(folder:GetChildren()) do
                    local hum = mob:FindFirstChildOfClass("Humanoid")
                    local mroot = mob:FindFirstChild("HumanoidRootPart")
                    if hum and hum.Health > 0 and mroot and (not farming or mobMatch(mob))
                        and (mroot.Position - root.Position).Magnitude <= radius then
                        if (mroot.Position - pos).Magnitude > 3 then
                            mroot.CFrame = point
                            mroot.AssemblyLinearVelocity = Vector3.zero
                        end
                        n += 1
                        if n >= 40 then break end
                    end
                end
            end)
        end
    end
end)

local function aliveNear(c, r, filter)
    local folder = workspace:FindFirstChild("Enemies")
    if not folder then return false end
    for _, mob in ipairs(folder:GetChildren()) do
        local hum = mob:FindFirstChildOfClass("Humanoid")
        local mr = mob:FindFirstChild("HumanoidRootPart")
        if hum and hum.Health > 0 and mr and (not filter or mob.Name == filter)
            and (mr.Position - c).Magnitude <= r then return true end
    end
    return false
end

        end

end)

-- Unstore fruits (unverified: LoadFruit remote and inventory format are from memory)
-- Only non-legendary, non-mythical fruits (Common / Uncommon / Rare)
local FallbackFruits = {
    "Rocket-Rocket", "Spin-Spin", "Blade-Blade", "Spring-Spring", "Bomb-Bomb", "Smoke-Smoke", "Spike-Spike",
    "Flame-Flame", "Ice-Ice", "Sand-Sand", "Dark-Dark", "Diamond-Diamond",
    "Light-Light", "Rubber-Rubber", "Ghost-Ghost", "Magma-Magma",
}
local AllowedNames = {}
for _, n in ipairs(FallbackFruits) do AllowedNames[n] = true end

local function fruitList()
    local names = {}
    for _, call in ipairs({ "getInventoryFruits", "getInventory" }) do
        local ok, res = pcall(function() return CommF:InvokeServer(call) end)
        if ok and type(res) == "table" then
            for _, item in pairs(res) do
                if type(item) == "table" then
                    local name = item.Name
                    local isFruit = item.Type == nil or item.Type == "Blox Fruit"
                    local rar = tonumber(item.Rarity)
                    if type(name) == "string" and isFruit then
                        if (rar and rar <= 2) or (not rar and AllowedNames[name]) then
                            table.insert(names, name)
                        end
                    end
                end
            end
            if #names > 0 then return names end
        end
    end
    return FallbackFruits
end

local function hasFruitTool(name)
    local char = lp.Character
    if char and char:FindFirstChild(name) then return true end
    return lp.Backpack:FindFirstChild(name) ~= nil
end

local lastUnstore = 0
local function unstoreFruits(force)
    if not force and os.clock() - lastUnstore < 30 then return end
    lastUnstore = os.clock()
    local taken = 0
    for _, name in ipairs(fruitList()) do
        if not hasFruitTool(name) then
            local ok, res = pcall(function() return CommF:InvokeServer("LoadFruit", name) end)
            if ok and res then taken += 1 end
            task.wait(0.4)
        end
    end
    return taken
end

-- Auto Raid (unverified: chip purchase, start button and island names are from memory)
local summonDetector
local function findSummon()
    if summonDetector and summonDetector.Parent then return summonDetector end
    local map = workspace:FindFirstChild("Map")
    if not map then return nil end
    for _, d in ipairs(map:GetDescendants()) do
        if d:IsA("ClickDetector") and d:GetFullName():find("RaidSummon", 1, true) then
            summonDetector = d
            return d
        end
    end
end

local function raidIslands()
    local loc = workspace:FindFirstChild("_WorldOrigin")
    loc = loc and loc:FindFirstChild("Locations")
    local list = {}
    if not loc then return list end
    for i = 5, 1, -1 do
        local isl = loc:FindFirstChild("Island " .. i)
        if isl then table.insert(list, isl) end
    end
    return list
end

local function hasChip()
    local char = lp.Character
    if char and char:FindFirstChild("Special Microchip") then return true end
    return lp.Backpack:FindFirstChild("Special Microchip") ~= nil
end

task.spawn(function()
    local lastStart, warned = 0, false
    local returnCF
    while true do
        task.wait(0.4)
        if Cfg.Raid and not Cfg.Farm then
            pcall(function()
                local root = getRoot()
                local folder = workspace:FindFirstChild("Enemies")
                if not (root and folder) then return end
                local islands = raidIslands()
                if #islands == 0 then
                    -- no raid running: unstore fruits, buy chip and press the start button
                    center = nil
                    if returnCF then
                        root.CFrame = returnCF
                        root.AssemblyLinearVelocity = Vector3.zero
                        returnCF = nil
                    end
                    if os.clock() - lastStart < 20 then return end
                    lastStart = os.clock()
                    if not hasChip() then
                        if Cfg.Unstore then unstoreFruits(false) end
                        CommF:InvokeServer("RaidsNpc", "Select", Cfg.RaidType)
                        task.wait(0.5)
                    end
                    local cd = findSummon()
                    if cd and fireclickdetector then
                        local part = cd.Parent
                        if part and part:IsA("BasePart") then
                            root.CFrame = part.CFrame * CFrame.new(0, 3, 6)
                            task.wait(0.4)
                        end
                        fireclickdetector(cd)
                        warned = false
                    elseif not warned then
                        warned = true
                        notify("Auto Raid", "Raid start button not found in this sea.")
                    end
                    return
                end
                if not returnCF then returnCF = root.CFrame end
                -- raid running: go for nearest mob, otherwise fly to the newest island
                if center and aliveNear(center, 40) then return end
                local best, bd = nil, 3000
                for _, mob in ipairs(folder:GetChildren()) do
                    local hum = mob:FindFirstChildOfClass("Humanoid")
                    local mr = mob:FindFirstChild("HumanoidRootPart")
                    if hum and hum.Health > 0 and mr then
                        local d = (mr.Position - root.Position).Magnitude
                        if d < bd then bd, best = d, mr end
                    end
                end
                if best then
                    center = best.Position
                else
                    local isl = islands[1]
                    local p = isl:IsA("BasePart") and isl.Position or (isl:IsA("Model") and isl:GetPivot().Position)
                    center = p
                end
            end)
        elseif not Cfg.Quest and not Cfg.Farm then
            if not Cfg.Raid then warned = false end
        end
    end
end)

-- Auto Farm Level (quests)
local Quests
pcall(function() Quests = require(RS:WaitForChild("Quests", 15)) end)

local function spawnFor(mob)
    local ok, res = pcall(function()
        local f = workspace:FindFirstChild("_WorldOrigin")
        f = f and f:FindFirstChild("EnemySpawns")
        if not f then return nil end
        for _, s in ipairs(f:GetChildren()) do
            if s.Name:find(mob, 1, true) then return s end
        end
    end)
    return ok and res or nil
end

local function seaRange()
    local pid = game.PlaceId
    if pid == 2753915549 then return 0, 699 end
    if pid == 4442272183 then return 700, 1499 end
    return 1500, math.huge
end

local function pickQuest()
    if not Quests then return nil end
    local lvl = lp:FindFirstChild("Data") and lp.Data:FindFirstChild("Level") and lp.Data.Level.Value or 1
    local lo, hi = seaRange()
    local best
    for key, levels in pairs(Quests) do
        if type(levels) == "table" then
            for idx, q in pairs(levels) do
                if type(q) == "table" and type(q.Task) == "table" and q.LevelReq
                    and q.LevelReq <= lvl and q.LevelReq >= lo and q.LevelReq <= hi then
                    local mob, cnt = next(q.Task)
                    if mob and type(cnt) == "number" and cnt > 1 then
                        if not best or q.LevelReq > best.req then
                            best = { key = key, idx = idx, mob = mob, req = q.LevelReq }
                        end
                    end
                end
            end
        end
    end
    return best
end

local function questTracked()
    local ok, vis = pcall(function()
        return lp.PlayerGui.Main.Quest.Visible
    end)
    return ok and vis
end

task.spawn(function()
    local quest, lastPick, lastAccept, warned = nil, 0, 0, false
    while true do
        task.wait(0.5)
        if Cfg.Quest and not Cfg.Farm and not Cfg.Raid then
            pcall(function()
                local root = getRoot()
                local folder = workspace:FindFirstChild("Enemies")
                if not (root and folder) then return end
                if not quest or os.clock() - lastPick > 30 then
                    lastPick = os.clock()
                    quest = pickQuest()
                end
                if not quest then
                    if not warned then warned = true notify("Auto Farm Level", "No suitable quest found for your level.") end
                    center = nil
                    questMob = nil
                    return
                end
                warned = false
                questMob = quest.mob
                if not questTracked() and os.clock() - lastAccept > 10 then
                    lastAccept = os.clock()
                    CommF:InvokeServer("StartQuest", quest.key, quest.idx)
                end
                if center and aliveNear(center, 40, quest.mob) then return end
                local best, bd = nil, math.huge
                for _, mob in ipairs(folder:GetChildren()) do
                    local hum = mob:FindFirstChildOfClass("Humanoid")
                    local mr = mob:FindFirstChild("HumanoidRootPart")
                    if mob.Name == quest.mob and hum and hum.Health > 0 and mr then
                        local d = (mr.Position - root.Position).Magnitude
                        if d < bd then bd, best = d, mr end
                    end
                end
                if best then
                    center = best.Position
                else
                    local s = spawnFor(quest.mob)
                    center = s and s.Position or nil
                end
            end)
        elseif not Cfg.Quest then
            questMob = nil
        end
    end
end)

-- flight (smooth, constant speed) above the pile
RunService.Heartbeat:Connect(function(dt)
    if not (active() and center) then return end
    local root = getRoot()
    if not root then return end
    local goal = center + Vector3.new(0, Cfg.Height, 0)
    local diff = goal - root.Position
    local dist = diff.Magnitude
    local step = Cfg.Speed * dt
    local newPos = dist <= step and goal or root.Position + diff.Unit * step
    root.CFrame = CFrame.lookAt(newPos, newPos - Vector3.new(0, 1, 0) + Vector3.new(0, 0, 0.001))
    root.AssemblyLinearVelocity = Vector3.zero
end)

-- noclip while farming
local lastClip = 0
local clipped = false
local function restoreCollide()
    clipped = false
    local char = lp.Character
    if not char then return end
    for _, p in ipairs(char:GetChildren()) do
        if p:IsA("BasePart") and (p.Name == "UpperTorso" or p.Name == "LowerTorso" or p.Name == "Torso" or p.Name == "Head") then
            p.CanCollide = true
        end
    end
end
RunService.Stepped:Connect(function()
    if not (active() and Cfg.NoClip and center) then
        if clipped then restoreCollide() end
        return
    end
    clipped = true
    if os.clock() - lastClip < 0.25 then return end
    lastClip = os.clock()
    local char = lp.Character
    if not char then return end
    for _, p in ipairs(char:GetDescendants()) do
        if p:IsA("BasePart") and p.CanCollide then p.CanCollide = false end
    end
end)
